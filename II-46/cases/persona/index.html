Paso 2: Crear DbHelper (Conexión y Comandos ADO.NET)

Centraliza el uso de SqlConnection, SqlCommand y SqlParameter para evitar código repetitivo y prevenir inyección SQL.

Data/DbHelper.vb 
Imports System.Configuration
Imports System.Data
Imports System.Data.SqlClient

Public Class DbHelper
    Private ReadOnly _connectionString As String

    Public Sub New()
        _connectionString = ConfigurationManager.ConnectionStrings("DefaultConnection").ConnectionString
    End Sub

    Public Function GetConnection() As SqlConnection
        Return New SqlConnection(_connectionString)
    End Function

    Public Function ExecuteNonQuery(query As String, Optional parameters As List(Of SqlParameter) = Nothing) As Integer
        Using conn As SqlConnection = GetConnection()
            Using cmd As New SqlCommand(query, conn)
                If parameters IsNot Nothing Then
                    cmd.Parameters.AddRange(parameters.ToArray())
                End If
                conn.Open()
                Return cmd.ExecuteNonQuery()
            End Using
        End Using
    End Function

    Public Function ExecuteQuery(query As String, Optional parameters As List(Of SqlParameter) = Nothing) As DataTable
        Dim dt As New DataTable()
        Using conn As SqlConnection = GetConnection()
            Using cmd As New SqlCommand(query, conn)
                If parameters IsNot Nothing Then
                    cmd.Parameters.AddRange(parameters.ToArray())
                End If
                Using adapter As New SqlDataAdapter(cmd)
                    adapter.Fill(dt)
                End Using
            End Using
        End Using
        Return dt
    End Function
End Class

Paso 3: Crear PersonaDB con métodos CRUD

Implementa GetAll, GetById, Insert, Update y Delete apoyándose en DbHelper.

Data/PersonaDB.vb 
Imports System.Data
Imports System.Data.SqlClient

Public Class PersonaDB
    Private ReadOnly _db As New DbHelper()

    Public Function GetAll() As List(Of Persona)
        Dim lista As New List(Of Persona)()
        Dim query As String = "SELECT IdPersona, Identificacion, Nombre, Apellidos, Correo, Edad FROM Personas ORDER BY IdPersona DESC"
        Dim dt As DataTable = _db.ExecuteQuery(query)

        For Each row As DataRow In dt.Rows
            lista.Add(MapRowToPersona(row))
        Next
        Return lista
    End Function

    Public Function GetById(id As Integer) As Persona
        Dim query As String = "SELECT IdPersona, Identificacion, Nombre, Apellidos, Correo, Edad FROM Personas WHERE IdPersona = @IdPersona"
        Dim params As New List(Of SqlParameter) From {
            New SqlParameter("@IdPersona", id)
        }
        Dim dt As DataTable = _db.ExecuteQuery(query, params)
        If dt.Rows.Count > 0 Then
            Return MapRowToPersona(dt.Rows(0))
        End If
        Return Nothing
    End Function

    Public Function Insert(p As Persona) As Boolean
        Dim query As String = "INSERT INTO Personas (Identificacion, Nombre, Apellidos, Correo, Edad) VALUES (@Identificacion, @Nombre, @Apellidos, @Correo, @Edad)"
        Dim params As New List(Of SqlParameter) From {
            New SqlParameter("@Identificacion", p.Identificacion),
            New SqlParameter("@Nombre", p.Nombre),
            New SqlParameter("@Apellidos", p.Apellidos),
            New SqlParameter("@Correo", p.Correo),
            New SqlParameter("@Edad", p.Edad)
        }
        Return _db.ExecuteNonQuery(query, params) > 0
    End Function

    Public Function Update(p As Persona) As Boolean
        Dim query As String = "UPDATE Personas SET Identificacion = @Identificacion, Nombre = @Nombre, Apellidos = @Apellidos, Correo = @Correo, Edad = @Edad WHERE IdPersona = @IdPersona"
        Dim params As New List(Of SqlParameter) From {
            New SqlParameter("@IdPersona", p.IdPersona),
            New SqlParameter("@Identificacion", p.Identificacion),
            New SqlParameter("@Nombre", p.Nombre),
            New SqlParameter("@Apellidos", p.Apellidos),
            New SqlParameter("@Correo", p.Correo),
            New SqlParameter("@Edad", p.Edad)
        }
        Return _db.ExecuteNonQuery(query, params) > 0
    End Function

    Public Function Delete(id As Integer) As Boolean
        Dim query As String = "DELETE FROM Personas WHERE IdPersona = @IdPersona"
        Dim params As New List(Of SqlParameter) From {
            New SqlParameter("@IdPersona", id)
        }
        Return _db.ExecuteNonQuery(query, params) > 0
    End Function

    Private Function MapRowToPersona(row As DataRow) As Persona
        Return New Persona() With {
            .IdPersona = Convert.ToInt32(row("IdPersona")),
            .Identificacion = row("Identificacion").ToString(),
            .Nombre = row("Nombre").ToString(),
            .Apellidos = row("Apellidos").ToString(),
            .Correo = row("Correo").ToString(),
            .Edad = Convert.ToInt32(row("Edad"))
        }
    End Function
End Class

Paso 4: Crear PersonaService (Reglas de Negocio y Validaciones)

Esta capa valida los datos antes de llamar a PersonaDB. No contiene código SQL ni controles de interfaz.

Services/PersonaService.vb 
Public Class PersonaService
    Private ReadOnly _personaDB As New PersonaDB()

    Public Function ListarPersonas() As List(Of Persona)
        Return _personaDB.GetAll()
    End Function

    Public Function ObtenerPorId(id As Integer) As Persona
        If id <= 0 Then
            Throw New ArgumentException("El ID de la persona no es válido.")
        End If
        Return _personaDB.GetById(id)
    End Function

    Public Sub GuardarPersona(p As Persona)
        ValidarPersona(p)
        If p.IdPersona = 0 Then
            _personaDB.Insert(p)
        Else
            _personaDB.Update(p)
        End If
    End Sub

    Public Sub EliminarPersona(id As Integer)
        If id <= 0 Then
            Throw New ArgumentException("Seleccione un registro válido para eliminar.")
        End If
        _personaDB.Delete(id)
    End Sub

    Private Sub ValidarPersona(p As Persona)
        If p Is Nothing Then Throw New ArgumentException("Datos requeridos.")
        If String.IsNullOrWhiteSpace(p.Identificacion) Then Throw New ArgumentException("La identificación es obligatoria.")
        If String.IsNullOrWhiteSpace(p.Nombre) Then Throw New ArgumentException("El nombre es obligatorio.")
        If String.IsNullOrWhiteSpace(p.Apellidos) Then Throw New ArgumentException("Los apellidos son obligatorios.")
        If String.IsNullOrWhiteSpace(p.Correo) OrElse Not p.Correo.Contains("@") Then Throw New ArgumentException("Ingrese un correo válido.")
        If p.Edad < 0 OrElse p.Edad > 120 Then Throw New ArgumentException("La edad debe estar entre 0 y 120 años.")
    End Sub
End Class

Personas.aspx.vb (Code-Behind) 
Public Class Personas
    Inherits System.Web.UI.Page

    Private ReadOnly _service As New PersonaService()

    Private Property IdPersonaEditar As Integer
        Get
            Return If(ViewState("IdPersonaEditar") IsNot Nothing, Convert.ToInt32(ViewState("IdPersonaEditar")), 0)
        End Get
        Set(value As Integer)
            ViewState("IdPersonaEditar") = value
        End Set
    End Property

    Protected Sub Page_Load(ByVal sender As Object, ByVal e As System.EventArgs) Handles Me.Load
        If Not IsPostBack Then
            CargarGrid()
        End If
    End Sub

    Private Sub CargarGrid()
        Try
            gvPersonas.DataSource = _service.ListarPersonas()
            gvPersonas.DataBind()
        Catch ex As Exception
            SwalUtils.ShowError(Me, "Error al cargar datos: " & ex.Message)
        End Try
    End Sub

    Protected Sub btnGuardar_Click(sender As Object, e As EventArgs)
        Try
            Dim edadValor As Integer = 0
            If Not Integer.TryParse(txtEdad.Text.Trim(), edadValor) Then
                SwalUtils.ShowWarning(Me, "Ingrese una edad numérica válida.")
                Return
            End If

            Dim p As New Persona() With {
                .IdPersona = Me.IdPersonaEditar,
                .Identificacion = txtIdentificacion.Text.Trim(),
                .Nombre = txtNombre.Text.Trim(),
                .Apellidos = txtApellidos.Text.Trim(),
                .Correo = txtCorreo.Text.Trim(),
                .Edad = edadValor
            }

            Dim esEdicion As Boolean = (Me.IdPersonaEditar > 0)
            _service.GuardarPersona(p)

            LimpiarFormulario()
            CargarGrid()
            SwalUtils.ShowSuccess(Me, If(esEdicion, "Registro actualizado.", "Registro insertado."))
        Catch ex As Exception
            SwalUtils.ShowError(Me, ex.Message)
        End Try
    End Sub

    Protected Sub gvPersonas_RowCommand(sender As Object, e As GridViewCommandEventArgs)
        If e.CommandName = "Seleccionar" OrElse e.CommandName = "Eliminar" Then
            Try
                Dim rowIndex As Integer = Convert.ToInt32(e.CommandArgument)
                Dim idPersona As Integer = Convert.ToInt32(gvPersonas.DataKeys(rowIndex).Value)

                If e.CommandName = "Seleccionar" Then
                    Dim p As Persona = _service.ObtenerPorId(idPersona)
                    If p IsNot Nothing Then
                        Me.IdPersonaEditar = p.IdPersona
                        txtIdentificacion.Text = p.Identificacion
                        txtNombre.Text = p.Nombre
                        txtApellidos.Text = p.Apellidos
                        txtCorreo.Text = p.Correo
                        txtEdad.Text = p.Edad.ToString()
                        lblTituloForm.Text = $"Editando Persona (ID: {p.IdPersona})"
                        btnGuardar.Text = "Actualizar"
                    End If
                ElseIf e.CommandName = "Eliminar" Then
                    _service.EliminarPersona(idPersona)
                    If Me.IdPersonaEditar = idPersona Then LimpiarFormulario()
                    CargarGrid()
                    SwalUtils.ShowSuccess(Me, "Registro eliminado correctamente.")
                End If
            Catch ex As Exception
                SwalUtils.ShowError(Me, ex.Message)
            End Try
        End If
    End Sub

    Protected Sub btnLimpiar_Click(sender As Object, e As EventArgs)
        LimpiarFormulario()
    End Sub

    Private Sub LimpiarFormulario()
        Me.IdPersonaEditar = 0
        txtIdentificacion.Text = ""
        txtNombre.Text = ""
        txtApellidos.Text = ""
        txtCorreo.Text = ""
        txtEdad.Text = ""
        lblTituloForm.Text = "Nueva Persona"
        btnGuardar.Text = "Guardar"
    End Sub
End Class

Paso 6: Usar SwalUtils en la capa UI

Clase utilitaria exclusiva para la capa de presentación que inyecta alertas de SweetAlert2 mediante ScriptManager.

Utils/SwalUtils.vb 
Imports System.Web.UI

Public Class SwalUtils
    Public Shared Sub ShowAlert(page As Page, title As String, message As String, icon As String)
        Dim cleanMsg As String = message.Replace("'", "\'").Replace(vbCrLf, " ")
        Dim script As String = $"Swal.fire({{ title: '{title}', text: '{cleanMsg}', icon: '{icon}' }});"
        ScriptManager.RegisterStartupScript(page, page.GetType(), Guid.NewGuid().ToString(), script, True)
    End Sub

    Public Shared Sub ShowSuccess(page As Page, message As String)
        ShowAlert(page, "¡Éxito!", message, "success")
    End Sub

    Public Shared Sub ShowError(page As Page, message As String)
        ShowAlert(page, "Error", message, "error")
    End Sub

    Public Shared Sub ShowWarning(page As Page, message As String)
        ShowAlert(page, "Atención", message, "warning")
    End Sub
End Class
